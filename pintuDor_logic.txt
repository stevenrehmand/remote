<?php
/**
 * AMETHYST PINTU LOGIC v8.0 - ACTIVE DIRECTORY SCANNER
 * Scan hanya pada folder yang sedang dibuka (Current Path).
 */

// 1. Tangkap direktori yang sedang dikunjungi
// Jika $currentPath tidak ada, gunakan folder tempat script berada
$dir = (isset($currentPath) && !empty($currentPath)) ? $currentPath : getcwd();

// Pastikan path valid
if (!is_dir($dir)) {
    echo "<div style='color:red;'>Error: Folder tidak valid atau tidak dapat diakses.</div>";
    return;
}

$found = [];

// --- DATABASE DANGER FUNCTIONS (SEMUA CRITICAL) ---
$f_critical = [
    'base64_decode', 'rawurldecode', 'urldecode', 'gzinflate', 'gzuncompress', 'str_rot13',
    'convert_uu', 'htmlspecialchars_decode', 'bin2hex', 'hex2bin', 'hexdec', 'chr', 'strrev',
    'goto', 'implode', 'strtr', 'extract', 'parse_str', 'substr', 'mb_substr', 'str_replace',
    'substr_replace', 'preg_replace', 'exif_read_data', 'readgzfile', 'eval', 'exec',
    'shell_exec', 'system', 'passthru', 'pcntl_fork', 'fsockopen', 'proc_open', 'popen',
    'assert', 'posix_kill', 'posix_getuid', 'getenv', 'phpinfo', 'mysqli_query', 'mysql_query',
    'file_put_contents', 'file_get_contents', 'move_uploaded_file', '$_files', 'copy',
    'include', 'require', 'mail', 'curl_init', 'symlink', '__halt_compiler',
    'fwrite', 'fopen', 'fputs', 'is_uploaded_file', '$_post', '$_get', '$_request',
    'html_file_input', 'html_form_enctype', 'upload_button_name'
];

// 2. Ambil daftar file di direktori aktif saja (bukan recursive)
$files = scandir($dir);

foreach ($files as $file) {
    $fPath = $dir . DIRECTORY_SEPARATOR . $file;
    
    // Scan hanya file PHP di folder aktif ini
    if (is_file($fPath) && pathinfo($file, PATHINFO_EXTENSION) == 'php' && $file != basename($_SERVER['PHP_SELF'])) {
        
        $content = @file_get_contents($fPath);
        if (empty($content)) continue;

        $score = 0; 
        $reasons = [];

        // Cek semua fungsi critical (Case Insensitive)
        foreach($f_critical as $f) { 
            if(stripos($content, $f) !== false) { 
                $score += 60; 
                $reasons[] = $f; 
            } 
        }

        // Heuristic Pattern (Kombinasi paling berbahaya)
        if (preg_match('/eval\s*\(\s*(base64_decode|gzinflate|\$_|str_rot13)/i', $content)) {
            $score += 100;
            $reasons[] = "BACKDOOR_PATTERN";
        }

        // Jika ditemukan kecurigaan (skor di atas 60)
        if ($score >= 60) {
            $found[] = [
                'name' => $file,
                'path' => $fPath,
                'score' => $score,
                'reasons' => array_unique($reasons)
            ];
        }
    }
}

// --- 3. GENERATE TAMPILAN HASIL ---
if (empty($found)) {
    $scan_result = "<div style='color:#00ff00; padding:10px; border:1px solid #00ff00; background:rgba(0,255,0,0.1);'>✅ Folder aktif aman.</div>";
} else {
    $html = "<div style='background:#111; padding:10px; border:1px solid #444;'>";
    $html .= "<h4 style='color:#ccc; margin-top:0;'>Scanner Result in: <span style='color:var(--accent);'>$dir</span></h4>";
    $html .= "<div style='overflow-x:auto;'><table style='width:100%; border-collapse:collapse; font-size:11px; font-family:monospace;'>";
    $html .= "<tr style='background:#222; color:#888;'><th style='padding:8px; text-align:left;'>FILENAME</th><th style='padding:8px; text-align:left;'>DETECTED FUNCTIONS</th><th style='padding:8px; text-align:right;'>SCORE</th></tr>";
    
    foreach ($found as $item) {
        $color = ($item['score'] >= 100) ? "#ff4d4d" : "#ffa500";
        $html .= "<tr style='border-bottom:1px solid #333;'>";
        $html .= "<td style='padding:8px;'><b style='color:$color;'>".$item['name']."</b></td>";
        $html .= "<td style='padding:8px; color:#999;'>".implode(", ", $item['reasons'])."</td>";
        $html .= "<td style='padding:8px; text-align:right; font-weight:bold; color:$color;'>".$item['score']."</td>";
        $html .= "</tr>";
    }
    $html .= "</table></div></div>";
    $scan_result = $html;
}

echo $scan_result;
?>
