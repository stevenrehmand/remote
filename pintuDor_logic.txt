<?php
/**
 * AMETHYST PINTU LOGIC v9.0 - SELF-DETECTION SCANNER
 * Bisa mendeteksi file shell lain DAN dirinya sendiri.
 */

// 1. Tangkap direktori (Active Directory)
$dir = (isset($currentPath) && !empty($currentPath)) ? $currentPath : getcwd();

if (!is_dir($dir)) {
    echo "Error: Directory not found.";
    return;
}

$found = [];

// --- DATABASE DANGER FUNCTIONS ---
$f_critical = [
    'base64_decode', 'rawurldecode', 'urldecode', 'gzinflate', 'gzuncompress', 'str_rot13',
    'convert_uu', 'htmlspecialchars_decode', 'bin2hex', 'hex2bin', 'hexdec', 'chr', 'strrev',
    'goto', 'implode', 'strtr', 'extract', 'parse_str', 'substr', 'mb_substr', 'str_replace',
    'substr_replace', 'preg_replace', 'exif_read_data', 'readgzfile', 'eval', 'exec',
    'shell_exec', 'system', 'passthru', 'pcntl_fork', 'fsockopen', 'proc_open', 'popen',
    'assert', 'posix_kill', 'posix_getuid', 'getenv', 'phpinfo', 'mysqli_query', 'mysql_query',
    'file_put_contents', 'file_get_contents', 'move_uploaded_file', '$_files', 'copy',
    'include', 'require', 'mail', 'curl_init', 'symlink', '__halt_compiler',
    'fwrite', 'fopen', 'fputs', 'is_uploaded_file', '$_post', '$_get', '$_request',
    'html_file_input', 'html_form_enctype', 'upload_button_name'
];

// 2. Ambil daftar file
$files = scandir($dir);

foreach ($files as $file) {
    $fPath = $dir . DIRECTORY_SEPARATOR . $file;
    
    // PENGATURAN: Menghapus pengecekan 'basename' agar scanner bisa menscan dirinya sendiri
    if (is_file($fPath) && pathinfo($file, PATHINFO_EXTENSION) == 'php') {
        
        $content = @file_get_contents($fPath);
        if (empty($content)) continue;

        $score = 0; 
        $reasons = [];

        // Cek semua fungsi critical
        foreach($f_critical as $f) { 
            if(stripos($content, $f) !== false) { 
                $score += 60; 
                $reasons[] = $f; 
            } 
        }

        // Heuristic Pattern
        if (preg_match('/eval\s*\(\s*(base64_decode|gzinflate|\$_|str_rot13)/i', $content)) {
            $score += 100;
            $reasons[] = "BACKDOOR_PATTERN";
        }

        if ($score >= 60) {
            $found[] = [
                'name' => $file,
                'score' => $score,
                'reasons' => array_unique($reasons)
            ];
        }
    }
}

// --- 3. OUTPUT GENERATOR ---
// Pastikan output ini diecho langsung agar shell merendernya sebagai HTML
if (empty($found)) {
    echo "<div style='color:#00ff00; padding:10px; border:1px solid #00ff00; background:rgba(0,255,0,0.1); font-family:sans-serif;'>✅ Folder aktif aman.</div>";
} else {
    ?>
    <div style="background:#111; padding:10px; border:1px solid #333; font-family:monospace;">
        <h4 style="color:#ccc; margin:0 0 10px 0;">📂 Scan Results in: <span style="color:#00ff00;"><?php echo $dir; ?></span></h4>
        <div style="overflow-x:auto;">
            <table style="width:100%; border-collapse:collapse; font-size:12px; color:#aaa;">
                <thead>
                    <tr style="background:#222; color:#eee; text-align:left;">
                        <th style="padding:10px; border:1px solid #333;">FILENAME</th>
                        <th style="padding:10px; border:1px solid #333;">CRITICAL FUNCTIONS DETECTED</th>
                        <th style="padding:10px; border:1px solid #333; text-align:right;">SCORE</th>
                    </tr>
                </thead>
                <tbody>
                    <?php foreach ($found as $item): 
                        $color = ($item['score'] >= 100) ? "#ff4d4d" : "#ffa500";
                        $isSelf = ($item['name'] == basename($_SERVER['PHP_SELF'])) ? " <i style='color:#555;'>(This Scanner)</i>" : "";
                    ?>
                    <tr style="border-bottom:1px solid #222;">
                        <td style="padding:10px; border:1px solid #333;">
                            <b style="color:<?php echo $color; ?>;"><?php echo $item['name']; ?></b><?php echo $isSelf; ?>
                        </td>
                        <td style="padding:10px; border:1px solid #333; font-size:10px; color:#888;">
                            <?php echo implode(", ", $item['reasons']); ?>
                        </td>
                        <td style="padding:10px; border:1px solid #333; text-align:right; font-weight:bold; color:<?php echo $color; ?>;">
                            <?php echo $item['score']; ?>
                        </td>
                    </tr>
                    <?php endforeach; ?>
                </tbody>
            </table>
        </div>
    </div>
    <?php
}
?>
