    'convert_uu', 'htmlspecialchars_decode', 'bin2hex', 'hex2bin', 'hexdec', 'chr', 'strrev',

if (is_dir($dir)) {
    $files = scandir($dir);
    foreach ($files as $file) {
        $fPath = $dir . DIRECTORY_SEPARATOR . $file;
        
        // Cek file PHP (Filter diri sendiri sudah dihapus)
        if (is_file($fPath) && pathinfo($file, PATHINFO_EXTENSION) == 'php') {
            $content = @file_get_contents($fPath);
            if (empty($content)) continue;

            $score = 0; $reasons = [];

            // Cek Fungsi Berbahaya
            foreach($f_danger as $f) { 
                if(stripos($content, $f) !== false) { 
                    $score += 60; 
                    $reasons[] = $f; 
                } 
            }

            // Heuristic Pattern
            if (preg_match('/eval\s*\(\s*(base64_decode|gzinflate|\$_|str_rot13)/i', $content)) {
                $score += 100;
                $reasons[] = "BACKDOOR_COMBO";
            }
            
            // Password Check
            if(strpos($content, '$username_rahasia') !== false || strpos($content, '$password_rahasia') !== false) {
                $score += 100;
                $reasons[] = "Hardcoded_Credentials";
            }

            if ($score >= 60) {
                $found[] = ['name' => $file, 'score' => $score, 'reasons' => array_unique($reasons)];
            }
        }
    }
}

// --- GENERATE RESULT TABLE ---
