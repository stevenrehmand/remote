<?php
/**
 * AMETHYST PINTU LOGIC v6.0 - MAXIMUM CRITICAL SCANNER
 * All-in-one Critical Database & Recursive Heuristic.
 */

// Konfigurasi Path
$dir = isset($currentPath) ? $currentPath : getcwd();
$found = [];

// --- DATABASE DANGER FUNCTIONS (ALL CRITICAL) ---
$f_critical = [
    // Original & Obfuscation
    'base64_decode', 'rawurldecode', 'urldecode', 'gzinflate', 'gzuncompress', 'str_rot13',
    'convert_uu', 'htmlspecialchars_decode', 'bin2hex', 'hex2bin', 'hexdec', 'chr', 'strrev',
    'goto', 'implode', 'strtr', 'extract', 'parse_str', 'substr', 'mb_substr', 'str_replace',
    'substr_replace', 'preg_replace', 'exif_read_data', 'readgzfile', 'eval', 'exec',
    'shell_exec', 'system', 'passthru', 'pcntl_fork', 'fsockopen', 'proc_open', 'popen',
    'assert', 'posix_kill', 'posix_getuid', 'getenv', 'phpinfo', 'mysqli_query', 'mysql_query',
    'file_put_contents', 'file_get_contents', 'move_uploaded_file', '$_files', 'copy',
    'include', 'require', 'mail', 'curl_init', 'symlink', '__halt_compiler',
    
    // TAMBAHAN UNTUK UPLOADER HUNTER
    'fwrite', 'fopen', 'fputs', 'is_uploaded_file', '$_post', '$_get', '$_request',
    'html_file_input', 'html_form_enctype', 'upload_button_name'
];

// --- PROSES SCANNING (RECURSIVE) ---
if (is_dir($dir)) {
    $iterator = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($dir));
    foreach ($iterator as $fileInfo) {
        if ($fileInfo->isDir()) continue;

        $fPath = $fileInfo->getPathname();
        $file  = $fileInfo->getFilename();
        $ext   = strtolower(pathinfo($file, PATHINFO_EXTENSION));

        // Cek file PHP dan variasinya
        $valid_ext = ['php', 'php7', 'php5', 'phtml', 'inc', 'phps'];
        if (in_array($ext, $valid_ext) && $file != basename($_SERVER['PHP_SELF'])) {
            
            $content = @file_get_contents($fPath);
            if (empty($content)) continue;

            $score = 0; 
            $reasons = [];

            // 1. CEK SEMUA FUNGSI CRITICAL (Case Insensitive)
            foreach($f_critical as $f) { 
                if(stripos($content, $f) !== false) { 
                    $score += 60; // Setiap temuan memberi skor tinggi
                    $reasons[] = "Critical:$f"; 
                } 
            }

            // 2. HEURISTIC PATTERN (Double Check untuk kombinasi mematikan)
            if (preg_match('/eval\s*\(\s*(base64_decode|gzinflate|str_rot13)/i', $content)) {
                $score += 100;
                $reasons[] = "Pattern:Backdoor_Combo";
            }

            // 3. DETEKSI DYNAMIC EXECUTION
            if (preg_match('/\$\_(GET|POST|REQUEST|COOKIE)\[.*\]\s*\(/i', $content)) {
                $score += 100;
                $reasons[] = "Pattern:Dynamic_Execution";
            }

            // 4. CEK VARIABLE PASSWORD
            if(preg_match('/(\$auth_pass|\$password|\$pass)\s*=/i', $content)) { 
                $score += 50; 
                $reasons[] = "Var:Password_Logic"; 
            }

            // Jika skor di atas ambang batas, masukkan ke daftar temuan
            if ($score >= 60) {
                $displayPath = str_replace($_SERVER['DOCUMENT_ROOT'], '', $fPath);
                $found[] = [
                    'name' => $file,
                    'full_path' => $fPath,
                    'rel_path' => $displayPath,
                    'score' => $score,
                    'reasons' => array_unique($reasons)
                ];
            }
        }
    }
}

// --- GENERATE RESULT TABLE ---
if (empty($found)) {
    $scan_result = "<div style='color:#00ff00; padding:10px;'>✅ Clean! Tidak ditemukan fungsi mencurigakan.</div>";
} else {
    $html = "<div style='overflow-x:auto;'><table style='width:100%; border-collapse:collapse; font-size:11px; font-family:monospace;'>";
    $html .= "<tr style='background:#222; color:#fff;'><th style='padding:8px; text-align:left;'>FILE & PATH</th><th style='padding:8px; text-align:left;'>CRITICAL DETECTED</th><th style='padding:8px; text-align:right;'>SCORE</th></tr>";
    
    foreach ($found as $item) {
        $color = ($item['score'] >= 120) ? "#ff4d4d" : "#ffa500";
        $html .= "<tr style='border-bottom:1px solid #333;'>";
        $html .= "<td style='padding:8px;'>
                    <b style='color:$color;'>".$item['name']."</b><br>
                    <small style='color:#666;'>".$item['rel_path']."</small>
                  </td>";
        $html .= "<td style='padding:8px; color:#aaa;'>".implode(", ", $item['reasons'])."</td>";
        $html .= "<td style='padding:8px; text-align:right; font-weight:bold; color:$color;'>".$item['score']."</td>";
        $html .= "</tr>";
    }
    $html .= "</table></div>";
    $scan_result = $html;
}

echo $scan_result;
?>
