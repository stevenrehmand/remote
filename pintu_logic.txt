<?php
/**
 * AMETHYST PINTU LOGIC v4.0 - ULTIMATE DATABASE
 * Database Danger Functions Terlengkap & Heuristic Scanner.
 */

$dir = $currentPath;
$files = scandir($dir);
$found = [];

// Mapping URL
$protocol = (isset($_SERVER['HTTPS']) && $_SERVER['HTTPS'] === 'on' ? "https" : "http");
$host = $_SERVER['HTTP_HOST'];
$docRoot = str_replace('\\', '/', $_SERVER['DOCUMENT_ROOT']);
$webPath = str_replace('\\', '/', $dir);
$relative = str_replace($docRoot, '', $webPath);
$baseUrl = $protocol . "://" . $host . $relative . "/";

// --- FULL DATABASE DANGER FUNCTIONS ---
$f_critical = ['eval', 'exec', 'shell_exec', 'system', 'passthru', 'pcntl_fork', 'fsockopen', 'proc_open', 'popen', 'assert', 'posix_kill', 'posix_setpgid', 'posix_setsid', 'posix_setuid', 'proc_nice', 'proc_close', 'proc_terminate', 'apache_child_terminate', 'create_function', 'include', 'include_once', 'require', 'require_once', '__halt_compiler'];

$f_obfus = ['base64_decode', 'rawurldecode', 'urldecode', 'gzinflate', 'gzuncompress', 'str_rot13', 'convert_uu', 'htmlspecialchars_decode', 'bin2hex', 'hex2bin', 'hexdec', 'chr', 'strrev', 'goto', 'implode', 'strtr', 'extract', 'parse_str', 'substr', 'mb_substr', 'str_replace', 'substr_replace', 'preg_replace', 'exif_read_data', 'readgzfile'];

$f_io_upload = ['fopen', 'fwrite', 'fputs', 'file_put_contents', 'file_get_contents', 'url_get_contents', 'stream_get_meta_data', 'move_uploaded_file', 'is_uploaded_file', 'copy', 'symlink', 'tmpfile', '$_FILES', '$_POST', '$_GET', '$_REQUEST'];

$f_info = ['posix_getuid', 'posix_geteuid', 'posix_getegid', 'posix_getpwuid', 'posix_getgrgid', 'posix_mkfifo', 'posix_getlogin', 'posix_ttyname', 'getenv', 'proc_get_status', 'get_cfg_var', 'disk_free_space', 'disk_total_space', 'diskfreespace', 'getlastmo', 'getmyinode', 'getmypid', 'getmyuid', 'getmygid', 'fileowner', 'filegroup', 'get_current_user', 'pathinfo', 'getcwd', 'sys_get_temp_dir', 'basename', 'phpinfo', '__FILE__'];

$f_misc = ['mysql_connect', 'mysqli_connect', 'mysqli_query', 'mysql_query', 'mail', 'putenv', 'curl_init', 'allow_url_fopen', 'ini_set', 'set_time_limit', 'session_start', 'error_reporting', 'get_magic_quotes_gpc'];

foreach ($files as $file) {
    $fPath = $dir . DIRECTORY_SEPARATOR . $file;
    if (is_file($fPath) && pathinfo($file, PATHINFO_EXTENSION) == 'php' && $file != basename($_SERVER['PHP_SELF'])) {
        $content = @file_get_contents($fPath);
        $score = 0; $reasons = [];

        foreach($f_critical as $f) { if(stripos($content, $f) !== false) { $score += 60; $reasons[] = "Critical:$f"; } }
        foreach($f_obfus as $f)    { if(stripos($content, $f) !== false) { $score += 25; $reasons[] = "Obfus:$f"; } }
        foreach($f_io_upload as $f){ if(stripos($content, $f) !== false) { $score += 30; $reasons[] = "I/O-Upload:$f"; } }
        foreach($f_info as $f)     { if(stripos($content, $f) !== false) { $score += 5; } }
        foreach($f_misc as $f)     { if(stripos($content, $f) !== false) { $score += 15; $reasons[] = "Misc:$f"; } }

        if(strpos($content, '$auth_pass') !== false || strpos($content, '$password') !== false) { $score += 50; $reasons[] = "Variable:Password"; }

        if ($score >= 45) {
            $found[] = ['name' => $file, 'score' => $score, 'reasons' => array_unique($reasons)];
        }
    }
}

// GENERATE RESULT TABLE
if (empty($found)) {
    $scan_result = "<div style='color:var(--success); padding:10px;'>✅ No suspicious files detected.</div>";
} else {
    $html = "<div style='overflow-x:auto;'><table style='width:100%; border-collapse:collapse; font-size:12px;'>";
    $html .= "<tr style='background:#000; color:#555;'><th style='padding:10px; text-align:left;'>FILE</th><th style='padding:10px; text-align:left;'>REASONS</th><th style='padding:10px; text-align:right;'>ACTIONS</th></tr>";
    foreach ($found as $item) {
        $color = ($item['score'] >= 100) ? "var(--danger)" : "var(--gold)";
        $fUrl = $baseUrl . $item['name'];
        $html .= "<tr style='border-bottom:1px solid rgba(255,255,255,0.05);'>";
        $html .= "<td style='padding:10px;'><b style='color:$color;'>".$item['name']."</b><br><small>Score: ".$item['score']."</small></td>";
        $html .= "<td style='padding:10px; color:#999;'>".implode(", ", $item['reasons'])."</td>";
        $html .= "<td style='padding:10px; text-align:right;'>
                    <a href='$fUrl' target='_blank' class='btn' style='padding:4px 8px; font-size:10px; background:var(--accent); border:none;'>VIEW URL</a>
                    <a href='?path=".urlencode($dir)."&delete=".urlencode($item['name'])."' onclick='return confirm(\"Delete?\")' class='btn' style='padding:4px 8px; font-size:10px; background:var(--danger); border:none;'>DEL</a>
                  </td>";
        $html .= "</tr>";
    }
    $html .= "</table></div>";
    $scan_result = $html;
}
?>
